# the assistant answers a question
until window.__app
fill [name="text"] Hello there
tap #send
until window.__app.items().some(m => m.role === 'assistant')
expect document.querySelector('[data-msg="assistant"]').textContent.includes('Hello there')

# the question and the answer both stay in the conversation
until window.__app
fill [name="text"] First question
tap #send
until window.__app.items().filter(m => m.role === 'assistant').length === 1
fill [name="text"] Second question
tap #send
until window.__app.items().filter(m => m.role === 'assistant').length === 2
expect window.__app.items().filter(m => m.role === 'user').length === 2
expect document.querySelectorAll('[data-msg="user"]').length === 2

# a new conversation clears the chat
until window.__app
tap #clear
until document.querySelector('.modal')
tap text:Clear
until window.__app.items().length === 0
expect document.querySelector('.empty') !== null
